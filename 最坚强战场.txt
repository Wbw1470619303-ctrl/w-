local function setCollision(enabled)
	local char = LocalPlayer.Character
	if char then
		for _, part in ipairs(char:GetDescendants()) do
			if part:IsA("BasePart") then
				part.CanCollide = enabled
			end
		end
	end
end

local function cleanupAntiFlingConstraints()
	local char = LocalPlayer.Character
	if char then
		local hrp = char:FindFirstChild("HumanoidRootPart")
		if hrp then
			local bv = hrp:FindFirstChild("AntiFlingBV")
			if bv then bv:Destroy() end
			local bg = hrp:FindFirstChild("AntiFlingBG")
			if bg then bg:Destroy() end
		end
	end
end

local function enableAntiFling()
	if antiFlingHeartbeat then
		antiFlingHeartbeat:Disconnect()
		antiFlingHeartbeat = nil
	end
	if antiFlingHeartbeat2 then
		antiFlingHeartbeat2:Disconnect()
		antiFlingHeartbeat2 = nil
	end
	cleanupAntiFlingConstraints()
	antiFlingEnabled = true
	AntiFlingToggle.BackgroundColor3 = Color3.fromRGB(0, 255, 0)
	setCollision(false)
	antiFlingHeartbeat = RunService.Stepped:Connect(function(_, deltaTime)
		if not antiFlingEnabled then return end
		local char = LocalPlayer.Character
		if not char then return end
		local hrp = char:FindFirstChild("HumanoidRootPart")
		local hum = char:FindFirstChild("Humanoid")
		if not hrp or not hum then return end
		if hum.PlatformStand then
			hum.PlatformStand = false
		end
		local bv = hrp:FindFirstChild("AntiFlingBV")
		if not bv then
			bv = Instance.new("BodyVelocity")
			bv.Name = "AntiFlingBV"
			bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
			bv.Velocity = Vector3.zero
			bv.Parent = hrp
		end
		local bg = hrp:FindFirstChild("AntiFlingBG")
		if not bg then
			bg = Instance.new("BodyGyro")
			bg.Name = "AntiFlingBG"
			bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
			bg.CFrame = hrp.CFrame
			bg.Parent = hrp
		end
		local moveDir = hum.MoveDirection
		local targetVel = Vector3.zero
		if moveDir.Magnitude > 0 then
			targetVel = moveDir * hum.WalkSpeed
		end
		hrp.AssemblyLinearVelocity = targetVel
		bv.Velocity = targetVel
		bg.CFrame = hrp.CFrame
	end)
	antiFlingHeartbeat2 = RunService.Heartbeat:Connect(function()
		if not antiFlingEnabled then return end
		local char = LocalPlayer.Character
		if not char then return end
		local hrp = char:FindFirstChild("HumanoidRootPart")
		local hum = char:FindFirstChild("Humanoid")
		if not hrp or not hum then return end
		local bv = hrp:FindFirstChild("AntiFlingBV")
		if not bv then return end
		local moveDir = hum.MoveDirection
		local targetVel = Vector3.zero
		if moveDir.Magnitude > 0 then
			targetVel = moveDir * hum.WalkSpeed
		end
		hrp.AssemblyLinearVelocity = targetVel
		bv.Velocity = targetVel
	end)
end

local function disableAntiFling()
	antiFlingEnabled = false
	AntiFlingToggle.BackgroundColor3 = Color3.fromRGB(30, 30, 40)
	if antiFlingHeartbeat then
		antiFlingHeartbeat:Disconnect()
		antiFlingHeartbeat = nil
	end
	if antiFlingHeartbeat2 then
		antiFlingHeartbeat2:Disconnect()
		antiFlingHeartbeat2 = nil
	end
	local char = LocalPlayer.Character
	if char then
		local hrp = char:FindFirstChild("HumanoidRootPart")
		if hrp then
			local bv = hrp:FindFirstChild("AntiFlingBV")
			if bv then bv:Destroy() end
			local bg = hrp:FindFirstChild("AntiFlingBG")
			if bg then bg:Destroy() end
		end
		local hum = char:FindFirstChild("Humanoid")
		if hum then
			hum.PlatformStand = false
		end
	end
	setCollision(true)
end

AntiFlingToggle.MouseButton1Click:Connect(function()
	if antiFlingEnabled then
		disableAntiFling()
	else
		enableAntiFling()
	end
end)

LocalPlayer.CharacterAdded:Connect(function(char)
	if antiFlingEnabled then
		task.wait(0.3)
		if antiFlingEnabled then
			enableAntiFling()
		end
	end
	char.DescendantAdded:Connect(function(descendant)
		if antiFlingEnabled and descendant:IsA("BasePart") then
			descendant.CanCollide = false
		end
	end)
end)

local function animateGlow()
	local tween = TweenService:Create(GlowGradient, TweenInfo.new(4, Enum.EasingStyle.Linear), {Rotation = 360})
	tween:Play()
	tween.Completed:Connect(function()
		GlowGradient.Rotation = 0
		animateGlow()
	end)
end

local function setupHover(btn, normalColor, hoverColor, textNormal, textHover)
	btn.MouseEnter:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = hoverColor}):Play()
		if textHover then TweenService:Create(btn, TweenInfo.new(0.15), {TextColor3 = textHover}):Play() end
	end)
	btn.MouseLeave:Connect(function()
		TweenService:Create(btn, TweenInfo.new(0.15), {BackgroundColor3 = normalColor}):Play()
		if textNormal then TweenService:Create(btn, TweenInfo.new(0.15), {TextColor3 = textNormal}):Play() end
	end)
end

local function updatePlayerList()
	for _, child in pairs(ListFrame:GetChildren()) do
		if child:IsA("TextButton") or child:IsA("TextLabel") then child:Destroy() end
	end
	local count = 0
	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= LocalPlayer then
			count += 1
			local btn = Instance.new("TextButton")
			btn.Name = player.Name
			btn.Size = UDim2.new(1, 0, 0, 32)
			btn.BackgroundColor3 = Color3.fromRGB(18, 18, 28)
			btn.BackgroundTransparency = 0.4
			btn.BorderSizePixel = 0
			btn.Text = "  " .. player.DisplayName .. "  [" .. player.Name .. "]"
			btn.TextColor3 = Color3.fromRGB(180, 180, 190)
			btn.TextSize = 12
			btn.Font = Enum.Font.Code
			btn.TextXAlignment = Enum.TextXAlignment.Left
			btn.Parent = ListFrame
			btn.MouseButton1Click:Connect(function()
				selectedPlayer = player
				for _, other in pairs(ListFrame:GetChildren()) do
					if other:IsA("TextButton") then
						other.BackgroundColor3 = Color3.fromRGB(18, 18, 28)
						other.TextColor3 = Color3.fromRGB(180, 180, 190)
					end
				end
				btn.BackgroundColor3 = Color3.fromRGB(0, 70, 55)
				btn.TextColor3 = Color3.fromRGB(0, 255, 200)
				StatusLabel.Text = "已锁定目标: " .. player.DisplayName
				StatusLabel.TextColor3 = Color3.fromRGB(0, 255, 200)
			end)
		end
	end
	if count == 0 then
		local empty = Instance.new("TextLabel")
		empty.Size = UDim2.new(1, 0, 0, 32)
		empty.BackgroundTransparency = 1
		empty.Text = "  [ 无其他玩家 ]"
		empty.TextColor3 = Color3.fromRGB(80, 80, 90)
		empty.TextSize = 12
		empty.Font = Enum.Font.Code
		empty.TextXAlignment = Enum.TextXAlignment.Left
		empty.Parent = ListFrame
	end
end
Players.PlayerAdded:Connect(updatePlayerList)
Players.PlayerRemoving:Connect(function(plr)
	if selectedPlayer == plr then
		selectedPlayer = nil
		if isTeleporting then stopTeleport() end
		if isFlinging then stopFling() end
		StatusLabel.Text = "目标玩家已断开连接"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
	end
	updatePlayerList()
end)

function startTeleport()
	if not selectedPlayer then
		StatusLabel.Text = "错误: 未选择目标玩家"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end
	if isFlinging then stopFling() end

	local target = selectedPlayer
	local localChar = LocalPlayer.Character
	if not localChar then return end
	local localHRP = localChar:FindFirstChild("HumanoidRootPart")
	if not localHRP then return end

	-- 保存原位
	lastPosition = localHRP.CFrame
	isTeleporting = true
	TeleportBtn.Text = "停止"
	TeleportBtn.BackgroundColor3 = Color3.fromRGB(80, 60, 0)
	StatusLabel.Text = "正在绝对锁定 -> " .. target.DisplayName
	StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 0)

	-- 清理防甩飞约束，避免冲突
	cleanupAntiFlingConstraints()
	cleanupTeleportConstraints()

	-- 等待目标角色就绪
	local targetChar = target.Character
	if not targetChar then
		StatusLabel.Text = "目标角色未加载"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		isTeleporting = false
		return
	end
	local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
	if not targetHRP then
		StatusLabel.Text = "目标HRP未加载"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		isTeleporting = false
		return
	end

	-- 计算初始锁定点
	local targetCF
	if isFrontMode then
		targetCF = CFrame.lookAt(
			targetHRP.Position + targetHRP.CFrame.LookVector * 4.5,
			targetHRP.Position
		)
	else
		targetCF = CFrame.lookAt(
			targetHRP.Position - targetHRP.CFrame.LookVector * 4.5,
			targetHRP.Position
		)
	end

	-- 一键瞬移到位（无过程）
	localHRP.CFrame = targetCF
	localHRP.AssemblyLinearVelocity = Vector3.zero
	localHRP.AssemblyAngularVelocity = Vector3.zero
	localHRP.RotVelocity = Vector3.zero

	-- 禁用碰撞，防止被目标/环境推开
	for _, part in ipairs(localChar:GetDescendants()) do
		if part:IsA("BasePart") then
			part.CanCollide = false
			part.Massless = true      -- 质量归零，确保物理模拟不会干扰
		end
	end

	-- 禁用Humanoid的自动旋转和站姿，防止动画干扰朝向
	local hum = localChar:FindFirstChild("Humanoid")
	if hum then
		hum.AutoRotate = false
		hum.PlatformStand = true
	end

	-- Heartbeat每帧纯CFrame锁定（无BodyPosition/BodyGyro）
	teleportConnection = RunService.Heartbeat:Connect(function()
		if not isTeleporting then return end
		if not target or not target.Parent then
			stopTeleport()
			StatusLabel.Text = "目标丢失，传送终止"
			StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
			selectedPlayer = nil
			return
		end

		local targetChar = target.Character
		if not targetChar then stopTeleport(); return end
		local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
		if not targetHRP then stopTeleport(); return end

		local myChar = LocalPlayer.Character
		if not myChar then stopTeleport(); return end
		local myHRP = myChar:FindFirstChild("HumanoidRootPart")
		if not myHRP then stopTeleport(); return end

		-- 实时计算跟随位置
		local newTargetCF
		if isFrontMode then
			newTargetCF = CFrame.lookAt(
				targetHRP.Position + targetHRP.CFrame.LookVector * 4.5,
				targetHRP.Position
			)
		else
			newTargetCF = CFrame.lookAt(
				targetHRP.Position - targetHRP.CFrame.LookVector * 4.5,
				targetHRP.Position
			)
		end

		-- 仅用CFrame强制覆盖，服务器因为网络所有权会完全接受
		myHRP.CFrame = newTargetCF

		-- 清零所有可能产生漂移的速度
		myHRP.AssemblyLinearVelocity = Vector3.zero
		myHRP.AssemblyAngularVelocity = Vector3.zero
		myHRP.RotVelocity = Vector3.zero

		-- 冻结其他主要身体部件（防止四肢自己甩动影响视觉）
		for _, partName in ipairs({"Head", "UpperTorso", "LowerTorso", "Torso",
								   "LeftUpperArm", "RightUpperArm", "LeftUpperLeg", "RightUpperLeg"}) do
			local part = myChar:FindFirstChild(partName)
			if part and part:IsA("BasePart") then
				part.AssemblyLinearVelocity = Vector3.zero
				part.AssemblyAngularVelocity = Vector3.zero
				part.RotVelocity = Vector3.zero
			end
		end
	end)
end

function stopTeleport()
	isTeleporting = false
	if teleportConnection then teleportConnection:Disconnect(); teleportConnection = nil end
	TeleportBtn.Text = "传送"
	TeleportBtn.BackgroundColor3 = Color3.fromRGB(0, 60, 45)
	StatusLabel.Text = "传送已停止"
	StatusLabel.TextColor3 = Color3.fromRGB(160, 160, 170)

	cleanupTeleportConstraints()

	local myChar = LocalPlayer.Character
	if myChar then
		local myHRP = myChar:FindFirstChild("HumanoidRootPart")
		if myHRP then
			-- 确保没有任何残留约束器（可能之前其他功能留下的）
			for _, obj in ipairs(myHRP:GetChildren()) do
				if obj:IsA("BodyPosition") or obj:IsA("BodyGyro") then
					obj:Destroy()
				end
			end
		end

		local hum = myChar:FindFirstChild("Humanoid")
		if hum then
			hum.AutoRotate = true
			hum.PlatformStand = false
		end

		-- 恢复碰撞和质量
		for _, part in ipairs(myChar:GetDescendants()) do
			if part:IsA("BasePart") then
				part.CanCollide = true
				part.Massless = false
			end
		end
	end
end

function startFling()
	if not selectedPlayer then
		StatusLabel.Text = "错误: 未选择目标玩家"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end
	if isTeleporting then stopTeleport() end
	local target = selectedPlayer
	local localChar = LocalPlayer.Character
	if not localChar then return end
	local localHRP = localChar:FindFirstChild("HumanoidRootPart")
	if not localHRP then return end
	lastPosition = localHRP.CFrame
	local hum = localChar:FindFirstChild("Humanoid")
	if hum then
		hum.AutoRotate = false
		hum.PlatformStand = true
	end
	isFlinging = true
	FlingBtn.Text = "停止甩飞"
	FlingBtn.BackgroundColor3 = Color3.fromRGB(90, 90, 0)
	StatusLabel.Text = "正在甩飞 -> " .. target.DisplayName
	StatusLabel.TextColor3 = Color3.fromRGB(255, 230, 0)

	local bv = Instance.new("BodyVelocity")
	bv.MaxForce = Vector3.new(1, 1, 1) * 1e7
	bv.Velocity = Vector3.zero
	bv.Parent = localHRP

	flingConnection = RunService.Heartbeat:Connect(function()
		if not isFlinging then return end
		local myChar = LocalPlayer.Character
		if not myChar then return end
		local myHRP = myChar:FindFirstChild("HumanoidRootPart")
		if not myHRP then return end
		local targetChar = target and target.Character
		if not targetChar then
			stopFling()
			StatusLabel.Text = "目标丢失，甩飞终止"
			StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
			return
		end
		local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
		if not targetHRP then return end

		myHRP.CFrame = targetHRP.CFrame * CFrame.new(0, 0.2, 0)
		myHRP.AssemblyLinearVelocity = Vector3.new(0, 5000, 0) + (targetHRP.CFrame.LookVector * 3000)
		bv.Velocity = myHRP.AssemblyLinearVelocity

		for _, partName in ipairs({"Head", "UpperTorso", "Torso"}) do
			local part = myChar:FindFirstChild(partName)
			if part and part:IsA("BasePart") then
				part.CFrame = targetHRP.CFrame
				part.AssemblyLinearVelocity = myHRP.AssemblyLinearVelocity
			end
		end
		task.wait(0.05)
		bv.Velocity = myHRP.AssemblyLinearVelocity
	end)
end

function stopFling()
	isFlinging = false
	if flingConnection then flingConnection:Disconnect(); flingConnection = nil end
	local myChar = LocalPlayer.Character
	if myChar then
		for _, v in ipairs(myChar:GetDescendants()) do
			if v:IsA("BodyVelocity") then v:Destroy() end
		end
		for _, partName in ipairs({"Head", "UpperTorso", "LowerTorso", "HumanoidRootPart", "Torso"}) do
			local part = myChar:FindFirstChild(partName)
			if part and part:IsA("BasePart") then
				part.AssemblyLinearVelocity = Vector3.zero
				part.AssemblyAngularVelocity = Vector3.zero
			end
		end
		local hum = myChar:FindFirstChild("Humanoid")
		if hum then
			hum.AutoRotate = true
			hum.PlatformStand = false
		end
	end
	FlingBtn.Text = "插入碰撞甩飞"
	FlingBtn.BackgroundColor3 = Color3.fromRGB(60, 60, 0)
	StatusLabel.Text = "甩飞已停止"
	StatusLabel.TextColor3 = Color3.fromRGB(160, 160, 170)
end

local function stopAndReturn()
	if not isTeleporting and not isFlinging and not antiSaitamaLooping then
		StatusLabel.Text = "无效操作: 当前未在传送或甩飞状态"
		StatusLabel.TextColor3 = Color3.fromRGB(255, 80, 80)
		return
	end
	if isTeleporting then stopTeleport() end
	if isFlinging then stopFling() end
	if antiSaitamaLooping then stopAntiSaitamaLoop() end
	local myChar = LocalPlayer.Character
	if myChar and lastPosition then
		local myHRP = myChar:FindFirstChild("HumanoidRootPart")
		if myHRP then
			myHRP.CFrame = lastPosition
			myHRP.AssemblyLinearVelocity = Vector3.zero
			StatusLabel.Text = "已停止并返回原位"
			StatusLabel.TextColor3 = Color3.fromRGB(0, 255, 200)
		end
	end
	lastPosition = nil
end

local function toggleUI()
	isOpen = not isOpen
	if isOpen then
		Content.Visible = true
		MainFrame.Size = UDim2.new(0, 290, 0, 540)
		ToggleBtn.Text = "▼"
		GlowFrame.Visible = true
	else
		Content.Visible = false
		MainFrame.Size = UDim2.new(0, 290, 0, 38)
		ToggleBtn.Text = "▲"
		GlowFrame.Visible = false
	end
end
ToggleBtn.MouseButton1Click:Connect(toggleUI)

local dragStart = nil
local frameStart = nil
local isDragging = false
TitleBar.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		isDragging = false
		dragStart = input.Position
		frameStart = MainFrame.Position
		local conn
		conn = input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then
				conn:Disconnect()
				if not isDragging then toggleUI() end
				dragStart = nil
			end
		end)
	end
end)
TitleBar.InputChanged:Connect(function(input)
	if dragStart and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		if delta.Magnitude > 4 then isDragging = true end
		if isDragging then
			MainFrame.Position = UDim2.new(
				frameStart.X.Scale, frameStart.X.Offset + delta.X,
				frameStart.Y.Scale, frameStart.Y.Offset + delta.Y
			)
		end
	end
end)

local function updateScale(delta)
	currentScale = math.clamp(currentScale + delta, 0.6, 1.3)
	UIScale.Scale = currentScale
	ScaleValue.Text = math.floor(currentScale * 100) .. "%"
end
ScaleMinus.MouseButton1Click:Connect(function() updateScale(-0.1) end)
ScalePlus.MouseButton1Click:Connect(function() updateScale(0.1) end)
